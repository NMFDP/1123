import React, { useState, useEffect } from "react";
import { View, Text, StyleSheet } from "react-native";
import { Accelerometer } from "expo-sensors";
import ConfettiCannon from "react-native-confetti-cannon"; // Import Confetti effect

export default function App() {
  // State to store the message and confetti trigger
  const [message, setMessage] = useState("Shake the device!");
  const [showConfetti, setShowConfetti] = useState(false);

  useEffect(() => {
    // Subscribe to accelerometer updates
    let subscription = Accelerometer.addListener(({ x, y, z }) => {
      const acceleration = Math.sqrt(x * x + y * y + z * z);

      // Check if the shake threshold is exceeded
      if (acceleration > 1.5) {
        setMessage("🎉 Happy Birthday! 🎊");
        setShowConfetti(true); // Trigger confetti

        // Reset message and confetti after 3 seconds
        setTimeout(() => {
          setMessage("Shake the device!");
          setShowConfetti(false);
        }, 3000);
      }
    });

    // Cleanup function to remove the listener when the component unmounts
    return () => subscription.remove();
  }, []);

  return (
    <View style={styles.container}>
      {/* Display the message */}
      <Text style={styles.text}>{message}</Text>

      {/* Show confetti when the device is shaken */}
      {showConfetti && <ConfettiCannon count={100} origin={{ x: 200, y: 0 }} />}
    </View>
  );
}

// Styles for the UI
const styles = StyleSheet.create({
  container: { 
    flex: 1,
    justifyContent: "center",
    alignItems: "center",
    backgroundColor: "#f8f8f8",
  },
  text: { 
    fontSize: 20,
    fontWeight: "bold",
    textAlign: "center",
  },
});
